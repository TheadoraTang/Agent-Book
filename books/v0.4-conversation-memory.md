      def chat(self, messages: list[dict[str, str]], temperature: float | None = None) -> str:
          if self.config.api_key in ["", "YOUR_API_KEY"]:
              raise RuntimeError("请先把 LLMConfig 里的 api_key 替换成真实密钥。")

          openai_client = self._get_openai_client()
          completion = openai_client.chat.completions.create(
              model=self.config.model_name,
              messages=messages,
              temperature=self.config.temperature if temperature is None else temperature,
          )
          return completion.choices[0].message.content or ""

      def _get_openai_client(self) -> object:
          if self.openai_client is not None:
              return self.openai_client

          self.openai_client = OpenAI(
              api_key=self.config.api_key,
              base_url=self.config.base_url,
              timeout=self.config.timeout_seconds,
          )
          return self.openai_client


  class ConversationMemory:
      def __init__(self, max_rounds: int = 5, messages: list[dict[str, object]] | None = None) -> None:
          self.max_rounds = max_rounds
          self.messages = messages or []

      def add_user_message(self, content: str) -> None:
          self.messages.append({"role": "user", "content": content})
          self.trim()

      def add_assistant_message(self, content: str) -> None:
          self.messages.append({"role": "assistant", "content": content})
          self.trim()

      def build_messages(self, system_prompt: str, user_input: str | None = None) -> list[dict[str, object]]:
          messages = [{"role": "system", "content": system_prompt}] + self.messages
          if user_input is not None:
              messages.append({"role": "user", "content": user_input})
          return messages

      def trim(self) -> None:
          self.messages = self.messages[-self.max_rounds * 2 :]
